# Versão original preservada — CTE única com produto cartesiano série × programa
# e OR no JOIN que impede uso de índice. Substituída por SQL_COMPONENTES_POR_TURMA.
_SQL_COMPONENTES_POR_TURMA_V1 = f"""
WITH tmpComponentePrograma AS (
    SELECT iif(pcc.cd_componente_curricular IS NOT NULL, pcc.cd_componente_curricular,
                cc.cd_componente_curricular) AS Codigo,
            iif(pcc.dc_componente_curricular IS NOT NULL, pcc.dc_componente_curricular,
                cc.dc_componente_curricular) AS Descricao,
            esc.tp_escola                    AS TipoEscola,
            dtt.qt_hora_duracao              AS TurnoTurma,
            serie_ensino.sg_resumida_serie   AS AnoTurma,
            te.an_letivo                     AS anoletivo,
            te.cd_turma_escola               AS TurmaCodigo,
            serie_turma_grade.cd_serie_grade AS CodigoSerieGrade,
            gcc.cd_grade                     AS CodigoGradeSerie,
            gcc.cd_componente_curricular     AS CodigoComponenteSerie,
            pgcc.cd_grade                    AS CodigoGradePrograma,
            pgcc.cd_componente_curricular    AS CodigoComponentePrograma
    FROM turma_escola (NOLOCK) te
        INNER JOIN escola (NOLOCK) esc ON te.cd_escola = esc.cd_escola
        -- Serie Ensino
        LEFT JOIN serie_turma_escola (NOLOCK)
            ON serie_turma_escola.cd_turma_escola = te.cd_turma_escola
        LEFT JOIN serie_turma_grade (NOLOCK)
            ON serie_turma_grade.cd_turma_escola = serie_turma_escola.cd_turma_escola
            AND serie_turma_grade.dt_fim IS NULL
        LEFT JOIN escola_grade (NOLOCK)
            ON serie_turma_grade.cd_escola_grade = escola_grade.cd_escola_grade
        LEFT JOIN grade (NOLOCK) ON escola_grade.cd_grade = grade.cd_grade
        LEFT JOIN grade_componente_curricular (NOLOCK) gcc ON gcc.cd_grade = grade.cd_grade
        LEFT JOIN componente_curricular (NOLOCK) cc
            ON cc.cd_componente_curricular = gcc.cd_componente_curricular
            AND cc.dt_cancelamento IS NULL
        LEFT JOIN serie_ensino (NOLOCK) ON grade.cd_serie_ensino = serie_ensino.cd_serie_ensino
        -- Programa
        LEFT JOIN tipo_programa (NOLOCK) tp ON te.cd_tipo_programa = tp.cd_tipo_programa
        LEFT JOIN turma_escola_grade_programa (NOLOCK) tegp ON tegp.cd_turma_escola = te.cd_turma_escola
        LEFT JOIN escola_grade (NOLOCK) teg ON teg.cd_escola_grade = tegp.cd_escola_grade
        LEFT JOIN grade (NOLOCK) pg ON pg.cd_grade = teg.cd_grade
        LEFT JOIN grade_componente_curricular (NOLOCK) pgcc ON pgcc.cd_grade = teg.cd_grade
        LEFT JOIN componente_curricular (NOLOCK) pcc
            ON pgcc.cd_componente_curricular = pcc.cd_componente_curricular
            AND pcc.dt_cancelamento IS NULL
        -- Turno
        INNER JOIN duracao_tipo_turno dtt
            ON te.cd_tipo_turno = dtt.cd_tipo_turno AND te.cd_duracao = dtt.cd_duracao
    WHERE te.st_turma_escola IN ('O', 'A', 'C', 'E')
      AND te.an_letivo = ?  -- param 1: ano_letivo
),
tmpComponentes AS (
    SELECT esc.tp_escola                    AS TipoEscola,
            dtt.qt_hora_duracao              AS TurnoTurma,
            serie_ensino.sg_resumida_serie   AS AnoTurma,
            turma_escola.an_letivo           AS anoletivo,
            turma_escola.cd_turma_escola     AS TurmaCodigo,
            turma_escola.cd_escola           AS CodigoEscola,
            serie_turma_grade.cd_serie_grade AS CodigoSerieGrade,
            grade.cd_grade                   AS CodigoGrade,
            turma_escola.dt_fim_turma        AS DataFimTurma
    FROM turma_escola (NOLOCK)
        INNER JOIN escola (NOLOCK) esc ON turma_escola.cd_escola = esc.cd_escola
        INNER JOIN serie_turma_escola (NOLOCK)
            ON serie_turma_escola.cd_turma_escola = turma_escola.cd_turma_escola
        INNER JOIN serie_turma_grade (NOLOCK)
            ON serie_turma_grade.cd_turma_escola = serie_turma_escola.cd_turma_escola
        INNER JOIN escola_grade (NOLOCK)
            ON serie_turma_grade.cd_escola_grade = escola_grade.cd_escola_grade
        INNER JOIN grade (NOLOCK) ON escola_grade.cd_grade = grade.cd_grade
        INNER JOIN serie_ensino (NOLOCK) ON grade.cd_serie_ensino = serie_ensino.cd_serie_ensino
        INNER JOIN duracao_tipo_turno dtt
            ON turma_escola.cd_tipo_turno = dtt.cd_tipo_turno
            AND turma_escola.cd_duracao = dtt.cd_duracao
    WHERE turma_escola.st_turma_escola IN ('O', 'A', 'C', 'E')
      AND turma_escola.an_letivo = ?  -- param 2: ano_letivo
)
SELECT DISTINCT * FROM (
    -- SME — parte 1: join por serie_grade (tmpComponentePrograma)
    SELECT DISTINCT tmp.Codigo, tmp.Descricao, tmp.TipoEscola, tmp.TurnoTurma,
                    tmp.AnoTurma, tmp.anoletivo, tmp.TurmaCodigo,
                    vsc.cd_registro_funcional AS Professor,
                    0 AS AtribuicaoExterna
    FROM tmpComponentePrograma tmp
    LEFT JOIN atribuicao_aula (NOLOCK) aa
        ON ((tmp.CodigoGradeSerie = aa.cd_grade
              AND tmp.CodigoComponenteSerie = aa.cd_componente_curricular
              AND aa.cd_serie_grade = tmp.CodigoSerieGrade)
            OR (tmp.CodigoGradePrograma = aa.cd_grade
                AND tmp.CodigoComponentePrograma = aa.cd_componente_curricular
                AND (tmp.CodigoSerieGrade IS NULL
                     OR tmp.CodigoSerieGrade = aa.cd_serie_grade)))
        AND aa.dt_cancelamento IS NULL
        AND (aa.cd_motivo_disponibilizacao <> {_MOTIVO_DISPONIBILIZACAO_ERRO_CADASTRO}
             OR aa.cd_motivo_disponibilizacao IS NULL)
        AND (aa.dt_disponibilizacao_aulas IS NULL
             OR aa.cd_motivo_disponibilizacao = {_MOTIVO_DISPONIBILIZACAO_FIM_ANO_LETIVO})
        AND aa.an_atribuicao = tmp.anoletivo
    LEFT JOIN v_cargo_base_cotic (NOLOCK) vcbc ON aa.cd_cargo_base_servidor = vcbc.cd_cargo_base_servidor
    LEFT JOIN v_servidor_cotic (NOLOCK) vsc ON vcbc.cd_servidor = vsc.cd_servidor

    UNION ALL

    -- Externo — parte 1: join por serie_grade (tmpComponentePrograma)
    SELECT DISTINCT tmp.Codigo, tmp.Descricao, tmp.TipoEscola, tmp.TurnoTurma,
                    tmp.AnoTurma, tmp.anoletivo, tmp.TurmaCodigo,
                    pe.cd_cpf_pessoa AS Professor,
                    1 AS AtribuicaoExterna
    FROM tmpComponentePrograma tmp
    LEFT JOIN atribuicao_externo ae (NOLOCK)
        ON ((tmp.CodigoGradeSerie = ae.cd_grade
              AND tmp.CodigoComponenteSerie = ae.cd_componente_curricular)
            OR (tmp.CodigoGradePrograma = ae.cd_grade
                AND tmp.CodigoComponentePrograma = ae.cd_componente_curricular))
        AND ae.dt_cancelamento IS NULL
        AND (ae.cd_motivo_disponibilizacao_externo <> 1
             OR ae.cd_motivo_disponibilizacao_externo IS NULL)
        AND ae.an_atribuicao = tmp.anoletivo
    LEFT JOIN contrato_externo ce (NOLOCK) ON ce.cd_contrato_externo = ae.cd_contrato_externo
    LEFT JOIN pessoa pe (NOLOCK) ON pe.cd_pessoa = ce.cd_pessoa
    WHERE tmp.TipoEscola IN {_TIPOS_ESCOLA_EXTERNOS}

    UNION ALL

    -- SME — parte 2: join por escola/grade (tmpComponentes)
    SELECT DISTINCT componente_curricular.cd_componente_curricular AS Codigo,
                    componente_curricular.dc_componente_curricular AS Descricao,
                    tmp.TipoEscola, tmp.TurnoTurma, tmp.AnoTurma, tmp.anoletivo, tmp.TurmaCodigo,
                    servidor.cd_registro_funcional AS Professor,
                    0 AS AtribuicaoExterna
    FROM tmpComponentes tmp
    INNER JOIN atribuicao_aula (NOLOCK)
        ON tmp.CodigoGrade = atribuicao_aula.cd_grade
        AND atribuicao_aula.an_atribuicao = tmp.anoletivo
        AND atribuicao_aula.cd_unidade_educacao = tmp.CodigoEscola
    INNER JOIN componente_curricular (NOLOCK)
        ON atribuicao_aula.cd_componente_curricular = componente_curricular.cd_componente_curricular
    INNER JOIN v_cargo_base_cotic cargoServidor (NOLOCK)
        ON cargoServidor.cd_cargo_base_servidor = atribuicao_aula.cd_cargo_base_servidor
    INNER JOIN v_servidor_cotic servidor (NOLOCK)
        ON servidor.cd_servidor = cargoServidor.cd_servidor
    WHERE atribuicao_aula.dt_disponibilizacao_aulas IS NOT NULL
      AND atribuicao_aula.dt_cancelamento IS NULL
      AND atribuicao_aula.cd_motivo_disponibilizacao <> {_MOTIVO_DISPONIBILIZACAO_ERRO_CADASTRO}
      AND (atribuicao_aula.cd_motivo_disponibilizacao = {_MOTIVO_DISPONIBILIZACAO_FIM_ANO_LETIVO}
           OR atribuicao_aula.cd_motivo_disponibilizacao IS NULL)
      AND COALESCE(atribuicao_aula.dt_disponibilizacao_aulas, tmp.DataFimTurma) >= tmp.DataFimTurma

    UNION ALL

    -- Externo — parte 2: join por escola/grade (tmpComponentes)
    SELECT DISTINCT componente_curricular.cd_componente_curricular AS Codigo,
                    componente_curricular.dc_componente_curricular AS Descricao,
                    tmp.TipoEscola, tmp.TurnoTurma, tmp.AnoTurma, tmp.anoletivo, tmp.TurmaCodigo,
                    pe.cd_cpf_pessoa AS Professor,
                    1 AS AtribuicaoExterna
    FROM tmpComponentes tmp
    INNER JOIN atribuicao_externo ae (NOLOCK)
        ON tmp.CodigoGrade = ae.cd_grade
        AND ae.an_atribuicao = tmp.anoletivo
        AND ae.cd_unidade_educacao = tmp.CodigoEscola
    INNER JOIN componente_curricular (NOLOCK)
        ON ae.cd_componente_curricular = componente_curricular.cd_componente_curricular
    INNER JOIN contrato_externo ce (NOLOCK) ON ce.cd_contrato_externo = ae.cd_contrato_externo
    INNER JOIN pessoa pe (NOLOCK) ON pe.cd_pessoa = ce.cd_pessoa
    WHERE ae.dt_disponibilizacao IS NOT NULL
      AND ae.dt_cancelamento IS NULL
      AND ae.cd_motivo_disponibilizacao_externo <> 1
      AND (ae.cd_motivo_disponibilizacao_externo = 3
           OR ae.cd_motivo_disponibilizacao_externo IS NULL)
      AND COALESCE(ae.dt_disponibilizacao, tmp.DataFimTurma) >= tmp.DataFimTurma
      AND tmp.TipoEscola IN {_TIPOS_ESCOLA_EXTERNOS}
) x
"""
